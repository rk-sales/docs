function switchScreen(screenId, btn) {
  document.querySelectorAll('.app-phone-screen > .app-screen').forEach(function(el) {
    el.classList.remove('active');
  });

  const nextScreen = document.getElementById(screenId);
  if (!nextScreen) return;
  nextScreen.classList.add('active');

  if (btn) {
    document.querySelectorAll('.app-phone-screen .app-tab').forEach(function(el) {
      el.classList.remove('active');
    });
    btn.classList.add('active');
  }

  const data = screenData[screenId];
  if (data) {
    const title = document.getElementById('dyn-title-left');
    const description = document.getElementById('dyn-desc-left');
    if (title) title.textContent = data.title;
    if (description) description.textContent = data.desc;

    const tiles = document.querySelectorAll('.app-dynamic-layout .app-tile');
    tiles.forEach(function(tile) {
      tile.classList.remove('animate');
      void tile.offsetWidth;
      tile.classList.add('animate');
    });
  }
}
