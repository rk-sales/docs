## Кто такие Мокка {: #about data-kicker="От редакции" }

<div class="mk-article">
<div class="mk-article__lede">
<p class="mk-article__text">
Мокка — омниканальный цифровой сервис для шоппинга. Он позволяет клиентам совершать покупки с моментальной оплатой частями, без первоначального взноса и по удобному графику. Мы поговорили с командой сервиса и разобрали, почему за последние годы к ним пришли более 17 миллионов уникальных пользователей — 62% из которых составляют поколение GenZ и миллениалы. Более 70% покупателей возвращаются в магазины-партнёры за повторной покупкой. Сегодня Мокка работает с 10 000 магазинов по всей стране — от одежды и обуви до путешествий и ювелирных изделий.
</p>

<script>
(function() {
  const page = document.querySelector('.mk-page');
  if (!page) return;

  /* ---------- FLIPBOOK ---------- */
  const book = page.querySelector('.mk-book');
  if (book) {
    const pages = book.querySelectorAll('.mk-book__page');
    const total = pages.length;
    const nextBtn = book.querySelector('[data-next]');
    const prevBtn = book.querySelector('[data-prev]');
    const counter = book.querySelector('[data-current]');
    let current = 0;

    function updateBook() {
      pages.forEach((p, i) => {
        p.classList.toggle('is-flipped', i < current);
      });
      counter.textContent = String(current + 1).padStart(2, '0');
      prevBtn.disabled = current === 0;
      nextBtn.disabled = current === total - 1;
    }

    nextBtn.addEventListener('click', () => {
      if (current < total - 1) { current++; updateBook(); }
    });
    prevBtn.addEventListener('click', () => {
      if (current > 0) { current--; updateBook(); }
    });
    updateBook();
  }

  /* ---------- CHART ---------- */
  const chart = page.querySelector('.mk-chart');
  if (chart) {
    const tabs = chart.querySelectorAll('.mk-tab');
    const chartBody = chart.querySelector('[data-chart]');
    const labelEl = chart.querySelector('[data-label]');
    const total = 12000;

    const terms = {
      1:  { label: '1 месяц',  monthly: 12000 },
      3:  { label: '3 месяца', monthly: 4000 },
      6:  { label: '6 месяцев', monthly: 2000 },
      12: { label: '12 месяцев', monthly: 1000 }
    };

    function renderChart(term) {
      const t = terms[term];
      const count = parseInt(term, 10);
      chartBody.innerHTML = '';
      for (let i = 0; i < count; i++) {
        const bar = document.createElement('div');
        bar.className = 'mk-chart__bar' + (i === 0 ? ' mk-chart__bar--first' : '');
        bar.innerHTML =
          '<span class="mk-chart__value">' + t.monthly.toLocaleString('ru-RU') + ' ₽</span>' +
          '<div class="mk-chart__bar-fill" style="height:' + (i === 0 ? '80%' : '55%') + '"></div>' +
          '<span class="mk-chart__label">' + (i === 0 ? '30 дней' : 'Мес. ' + (i + 1)) + '</span>';
        chartBody.appendChild(bar);
      }
      labelEl.textContent = 'Заказ ' + total.toLocaleString('ru-RU') + ' ₽ · ' + t.label;
    }

    tabs.forEach(tab => {
      tab.addEventListener('click', () => {
        tabs.forEach(t => t.classList.toggle('mk-tab--active', t === tab));
        renderChart(tab.dataset.term);
      });
    });

    renderChart('3');
  }
})();
</script>
