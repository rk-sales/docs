<script>
(function () {
  const page = document.querySelector('.ym-page');
  if (!page) return;

  const sourceBtns = page.querySelectorAll('[data-source-switch] [data-source]');
  const balanceEl = page.querySelector('[data-wallet-balance]');
  const subEl = page.querySelector('[data-wallet-sub]');

  const data = {
    wallet: {
      balance: '2 450 ₽',
      sub: 'Доступно для оплаты'
    },
    card: {
      balance: '•••• 4276',
      sub: 'Привязанная карта · МИР'
    }
  };

  function setSource(source) {
    sourceBtns.forEach(b => b.classList.toggle('ym-source__btn--active', b.dataset.source === source));
    const d = data[source];
    if (balanceEl) balanceEl.textContent = d.balance;
    if (subEl) subEl.textContent = d.sub;
  }

  sourceBtns.forEach(btn => {
    btn.addEventListener('click', () => setSource(btn.dataset.source));
  });
})();
</script>
