<!-- Скрипт калькулятора -->
<script>
document.addEventListener('DOMContentLoaded', function() {
  const totalEl = document.getElementById('ro-total-value');
  if (!totalEl) return;

  const inputs = document.querySelectorAll('.ro-matrix input[type="radio"], .ro-matrix input[type="checkbox"]');

  function calculate() {
    let total = 0;
    
    // Срок аренды (обязательно)
    const months = document.querySelector('input[name="ro-months"]:checked');
    if (months) total += parseInt(months.value, 10);

    // Фискальный накопитель (обязательно)
    const fn = document.querySelector('input[name="ro-fn"]:checked');
    if (fn) total += parseInt(fn.value, 10);

    // Доп услуги (чекбоксы)
    const extras = document.querySelectorAll('input[name="ro-extra"]:checked');
    extras.forEach(function(extra) {
      total += parseInt(extra.value, 10);
    });

    // Форматирование с пробелами
    totalEl.textContent = total.toString().replace(/\B(?=(\d{3})+(?!\d))/g, " ") + " ₽";
  }

  // Запуск при изменении любого инпута
  inputs.forEach(function(input) {
    input.addEventListener('change', calculate);
  });

  // Первый расчет при загрузке (учитываем checked по умолчанию)
  calculate();
});
</script>
